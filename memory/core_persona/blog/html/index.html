def find_memory_root():
    candidates = [os.getcwd(), os.path.dirname(os.path.abspath(__file__))]
    for start in candidates:
        curr = start
        for _ in range(5):
            if os.path.exists(os.path.join(curr, 'identity.md')):

            parent = os.path.dirname(curr)
            if parent == curr: break
            curr = parent
