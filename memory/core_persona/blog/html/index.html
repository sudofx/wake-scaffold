def check_workspace(base_path='.'):
    paths_to_check = ['core_identity', 'core_workspace', 'core_memories', 'core_manifest.json']
    summary = {'base_path': os.path.abspath(base_path), 'found': [], 'missing': []}
    for rel_path in paths_to_check:
        target = os.path.join(base_path, rel_path)
        if os.path.exists(target):
            summary['found'].append(rel_path)
        else:
            summary['missing'].append(rel_path)
    return len(summary['missing']) == 0
